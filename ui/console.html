Custom agents

Your own agent reports each step to the local recorder.

  1. POST /sessions: task, agent name, passport.
  2. POST /sessions/{id}/events: navigate, form_submit, download or secret.
  3. POST /sessions/{id}/stop: seals the receipt.
curl and Python 
# the token is read from its file, never typed or printed
T=$(cat ~/.hashcam/recorder-token)
curl -s -H "X-HashCam-Token: $T" -H "Content-Type: application/json" \
  -d '{"task":"Buy one notebook on shop.test","agent":"my-agent","passport":{"hosts":["shop.test"],"minutes":30}}' \
  http://127.0.0.1:8770/sessions
import json, pathlib, urllib.request
TOKEN = (pathlib.Path.home() / ".hashcam" / "recorder-token").read_text().strip()

def call(path, body):
    req = urllib.request.Request("http://127.0.0.1:8770" + path, data=json.dumps(body).encode(),
        headers={"X-HashCam-Token": TOKEN, "Content-Type": "application/json"})
    return json.load(urllib.request.urlopen(req))

s = call("/sessions", {"task": "Buy one notebook on shop.test", "agent": "my-agent",
                       "passport": {"hosts": ["shop.test"], "minutes": 30}})
step = call(f"/sessions/{s['session_id']}/events", {"kind": "navigate", "url": "https://shop.test/"})
if step["decision"] != "ALLOW":
    raise SystemExit("not allowed: " + ", ".join(step["reasons"]))
